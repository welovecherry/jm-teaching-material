**75-100분 권한 매트릭스와 예외를 코드로 구현**

def check_policy(dept, system, requested_level, matrix):

allowed = matrix.get(dept, {}).get(system)

if allowed is None:

return False, '매트릭스에 정의되지 않은 접근'

levels = \['조회', '수정', '관리자'\]

if levels.index(requested_level) \> levels.index(allowed):

return False, f'허용 수준({allowed}) 초과 요청'

return True, '정책 준수'

*\[확인질문\] '조회' 권한만 허용된 부서가 '관리자' 권한을 요청하면 이 함수는 어떻게 판단할까요?*

**100-120분 정책+예외 통합 판단 로직**

def evaluate_access(user, dept, system, level, matrix, exceptions):

ok, reason = check_policy(dept, system, level, matrix)

if ok: return True, reason

for exc in exceptions:

if exc\['user'\]==user and exc\['system'\]==system and is_exception_valid(exc):

return True, '예외 승인으로 허용'

return False, reason

def check_sla_breach(requests):

now = datetime.now()

breached = \[\]

for r in requests:

if r\['status'\] == 'reviewing':

if elapsed \> r\['sla_hours'\]:

return breached
